!line /*
!line  * C Converted Whetstone Double Precision Benchmark
!line  *		Version 1.2	22 March 1998
!line  *
!line  *	(c) Copyright 1998 Painter Engineering, Inc.
!line  *		All Rights Reserved.
!line  *
!line  *		Permission is granted to use, duplicate, and
!line  *		publish this text and program as long as it
!line  *		includes this entire comment block and limited
!line  *		rights reference.
!line  *
!line  * Converted by Rich Painter, Painter Engineering, Inc. based on the
!line  * www.netlib.org benchmark/whetstoned version obtained 16 March 1998.
!line  *
!line  * A novel approach was used here to keep the look and feel of the
!line  * FORTRAN version.  Altering the FORTRAN-based array indices,
!line  * starting at element 1, to start at element 0 for C, would require
!line  * numerous changes, including decrementing the variable indices by 1.
!line  * Instead, the array E1[] was declared 1 element larger in C.  This
!line  * allows the FORTRAN index range to function without any literal or
!line  * variable indices changes.  The array element E1[0] is simply never
!line  * used and does not alter the benchmark results.
!line  *
!line  * The major FORTRAN comment blocks were retained to minimize
!line  * differences between versions.  Modules N5 and N12, like in the
!line  * FORTRAN version, have been eliminated here.
!line  *
!line  * An optional command-line argument has been provided [-c] to
!line  * offer continuous repetition of the entire benchmark.
!line  * An optional argument for setting an alternate LOOP count is also
!line  * provided.  Define PRINTOUT to cause the POUT() function to print
!line  * outputs at various stages.  Final timing measurements should be
!line  * made with the PRINTOUT undefined.
!line  *
!line  * Questions and comments may be directed to the author at
!line  *			r.painter@ieee.org
!line  */
!line /* The double precision Whetstone of netlib's whetstone.c (Painter, 1998,
!line  * after the FORTRAN A001), modules 1-11 with the same arithmetic, in Sixth
!line  * Edition C: no long (LOOP = 10, every count fits an int), += for +=,
!line  * timed by the shell's time. 10 loops = 1,000,000 Whetstone instructions. */
!line double T, T1, T2, E1[5];
!line int J, K, L;
!line double sin(), cos(), atan(), log(), exp(), sqrt();
!line
!line main()
!line {
!line 	int I, N1, N2, N3, N4, N6, N7, N8, N9, N10, N11, LOOP, R;
!line 	double X1, X2, X3, X4, X, Y, Z;
!line
!line 	T = .499975; T1 = 0.50025; T2 = 2.0;
!line 	LOOP = 30;
!line 	N1 = 0; N2 = 12*LOOP; N3 = 14*LOOP; N4 = 345*LOOP; N6 = 210*LOOP;
!line 	N7 = 32*LOOP; N8 = 899*LOOP; N9 = 616*LOOP; N10 = 0; N11 = 93*LOOP;
!line
!line 	X1 = 1.0; X2 = -1.0; X3 = -1.0; X4 = -1.0;
!line 	for (I = 1; I <= N1; I++) {
!line 		X1 = (X1 + X2 + X3 - X4) * T;
!line 		X2 = (X1 + X2 - X3 + X4) * T;
!line 		X3 = (X1 - X2 + X3 + X4) * T;
!line 		X4 = (-X1+ X2 + X3 + X4) * T;
!line 	}
!line 	E1[1] = 1.0; E1[2] = -1.0; E1[3] = -1.0; E1[4] = -1.0;
!line 	for (I = 1; I <= N2; I++) {
!line 		E1[1] = ( E1[1] + E1[2] + E1[3] - E1[4]) * T;
!line 		E1[2] = ( E1[1] + E1[2] - E1[3] + E1[4]) * T;
!line 		E1[3] = ( E1[1] - E1[2] + E1[3] + E1[4]) * T;
!line 		E1[4] = (-E1[1] + E1[2] + E1[3] + E1[4]) * T;
!line 	}
!line 	for (I = 1; I <= N3; I++)
!line 		PA(E1);
!line 	J = 1;
!line 	for (I = 1; I <= N4; I++) {
!line 		if (J == 1) J = 2; else J = 3;
!line 		if (J > 2) J = 0; else J = 1;
!line 		if (J < 1) J = 1; else J = 0;
!line 	}
!line 	J = 1; K = 2; L = 3;
!line 	for (I = 1; I <= N6; I++) {
!line 		J = J * (K-J) * (L-K);
!line 		K = L * K - (L-J) * K;
!line 		L = (L-K) * (K+J);
!line 		E1[L-1] = J + K + L;
!line 		E1[K-1] = J * K * L;
!line 	}
!line 	X = 0.5; Y = 0.5;
!line 	for (I = 1; I <= N7; I++) {
!line 		X = T * atan(T2*sin(X)*cos(X)/(cos(X+Y)+cos(X-Y)-1.0));
!line 		Y = T * atan(T2*sin(Y)*cos(Y)/(cos(X+Y)+cos(X-Y)-1.0));
!line 	}
!line 	X = 1.0; Y = 1.0; Z = 1.0;
!line 	for (I = 1; I <= N8; I++)
!line 		P3(X, Y, &Z);
!line 	J = 1; K = 2; L = 3; E1[1] = 1.0; E1[2] = 2.0; E1[3] = 3.0;
!line 	for (I = 1; I <= N9; I++)
!line 		P0();
!line 	J = 2; K = 3;
!line 	for (I = 1; I <= N10; I++) {
!line 		J = J + K; K = J + K; J = K - J; K = K - J - J;
!line 	}
!line 	X = 0.75;
!line 	for (I = 1; I <= N11; I++)
!line 		X = sqrt(exp(log(X)/T1));
!line 	/* V6 printf has no prototype: explicitly convert the double for %d. */
!line 	R = X*10000.0;
!line 	printf("3000 KWhetstones done; X = %d/10000\n", R);
!line }
!line
!line PA(E)
!line double E[];
!line {
!line 	J = 0;
!line L10:
!line 	E[1] = ( E[1] + E[2] + E[3] - E[4]) * T;
!line 	E[2] = ( E[1] + E[2] - E[3] + E[4]) * T;
!line 	E[3] = ( E[1] - E[2] + E[3] + E[4]) * T;
!line 	E[4] = (-E[1] + E[2] + E[3] + E[4]) / T2;
!line 	J += 1;
!line 	if (J < 6)
!line 		goto L10;
!line }
!line
!line P0()
!line {
!line 	E1[J] = E1[K];
!line 	E1[K] = E1[L];
!line 	E1[L] = E1[J];
!line }
!line
!line P3(X, Y, Z)
!line double X, Y, *Z;
!line {
!line 	double X1, Y1;
!line 	X1 = X; Y1 = Y;
!line 	X1 = T * (X1 + Y1);
!line 	Y1 = T * (X1 + Y1);
!line 	*Z = (X1 + Y1) / T2;
!line }
!eof
!wait 0.2
cc -O whet.c -lm
mv a.out whet
time ./whet
time ./whet
time ./whet
!raw cat >sieve.c
!line
!line /* Eratosthenes Sieve Prime Number Program in C, J. Gilbreath, BYTE,
!line  * September 1981 (the benchmark's published form; 1899 primes). */
!line
!line char flags[8190+1];
!line
!line main()
!line {
!line 	int i, prime, k, count, iter;
!line
!line 	printf("100 iterations\n");
!line 	for (iter = 1; iter <= 100; iter++) {
!line 		count = 0;
!line 		for (i = 0; i <= 8190; i++)
!line 			flags[i] = 1;
!line 		for (i = 0; i <= 8190; i++) {
!line 			if (flags[i]) {
!line 				prime = i + i + 3;
!line 				for (k = i + prime; k <= 8190; k += prime)
!line 					flags[k] = 0;
!line 				count++;
!line 			}
!line 		}
!line 	}
!line 	printf("\n%d primes\n", count);
!line }
!eof
!wait 0.2
cc -O sieve.c
mv a.out sieve
time ./sieve
time ./sieve
time ./sieve
